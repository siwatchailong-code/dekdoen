import "jsr:@supabase/functions-js/edge-runtime.d.ts"; import { createClient } from "npm:@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", "Content-Type": "application/json; charset=utf-8", }; function json(data: unknown, status = 200) { return new Response(JSON.stringify(data), { status, headers: corsHeaders, }); } // สร้างรหัสผ่านใหม่ function generatePassword(length = 10) { const chars = "ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz23456789"; const bytes = new Uint32Array(length); crypto.getRandomValues(bytes); return Array.from( bytes, (value) => chars[value % chars.length], ).join(""); } Deno.serve(async (req) => { // CORS if (req.method === "OPTIONS") { return new Response("ok", { status: 200, headers: corsHeaders, }); } if (req.method !== "POST") { return json( { ok: false, error: "Method not allowed", }, 405, ); } try { const supabaseUrl = Deno.env.get("SUPABASE_URL"); const serviceRoleKey = Deno.env.get( "SUPABASE_SERVICE_ROLE_KEY", ); if (!supabaseUrl || !serviceRoleKey) { return json( { ok: false, error: "Server ยังไม่ได้ตั้ง SUPABASE_URL หรือ SUPABASE_SERVICE_ROLE_KEY", }, 500, ); } // ========================================== // รับ Access Token ของแอดมินจาก Frontend // ========================================== const authHeader = req.headers.get("Authorization") || ""; if (!authHeader.startsWith("Bearer ")) { return json( { ok: false, error: "ไม่พบ token สำหรับตรวจสอบสิทธิ์", }, 401, ); } const accessToken = authHeader .replace("Bearer ", "") .trim(); // ========================================== // Client สำหรับงานสิทธิ์สูง // ใช้เฉพาะใน Edge Function // ========================================== const adminClient = createClient( supabaseUrl, serviceRoleKey, { auth: { autoRefreshToken: false, persistSession: false, }, }, ); // ========================================== // ตรวจสอบว่า token ใช้งานได้จริง // ========================================== const { data: { user: adminUser }, error: authError, } = await adminClient.auth.getUser(accessToken); if (authError || !adminUser) { console.error( "auth getUser error:", authError, ); return json( { ok: false, error: "เซสชันหมดอายุหรือไม่ถูกต้อง กรุณาเข้าสู่ระบบใหม่", }, 401, ); } // ========================================== // ตรวจสิทธิ์ Admin จาก profiles // รองรับระบบเดิมที่เก็บ role ไว้ใน DB // ========================================== const { data: adminProfile, error: profileError, } = await adminClient .from("profiles") .select("id, role, banned") .eq("id", adminUser.id) .maybeSingle(); if (profileError) { console.error( "admin profile check error:", profileError, ); return json( { ok: false, error: "ตรวจสอบสิทธิ์ผู้ดูแลระบบไม่สำเร็จ: " + profileError.message, }, 500, ); } const metadataRole = String( adminUser.app_metadata?.role || adminUser.user_metadata?.role || "", ).toLowerCase(); const isAdmin = adminProfile?.role === "admin" || metadataRole === "admin" || adminUser.app_metadata?.is_admin === true || adminUser.user_metadata?.is_admin === true; if (!isAdmin) { return json( { ok: false, error: "บัญชีนี้ไม่มีสิทธิ์ผู้ดูแลระบบ", }, 403, ); } // ========================================== // รับข้อมูล // ========================================== const body = await req.json().catch(() => ({})); let action = String(body?.action || "") .trim() .toLowerCase(); const targetId = String( body?.targetId || "", ).trim(); if (!action) { return json( { ok: false, error: "กรุณาระบุ action", }, 400, ); } if (!targetId) { return json( { ok: false, error: "กรุณาระบุ targetId", }, 400, ); } // ========================================== // รองรับชื่อ action จาก Frontend เดิมทั้งหมด // ========================================== if ( action === "delete" || action === "delete_user" || action === "deleteuser" ) { action = "delete_user"; } if ( action === "reset" || action === "reset_password" || action === "resetpassword" ) { action = "reset_password"; } // ========================================== // ห้ามลบบัญชีตัวเอง // ========================================== if ( action === "delete_user" && targetId === adminUser.id ) { return json( { ok: false, error: "ไม่สามารถลบบัญชีแอดมินที่กำลังใช้งานอยู่ได้", }, 400, ); } // ========================================== // ตรวจว่าบัญชีเป้าหมายมีจริง // ========================================== const { data: targetUser, error: targetError, } = await adminClient.auth.admin.getUserById( targetId, ); if (targetError || !targetUser?.user) { return json( { ok: false, error: targetError?.message || "ไม่พบบัญชีที่ต้องการจัดการ", }, 404, ); } // ========================================== // รีเซ็ตรหัสผ่าน // ถ้า Frontend ไม่ส่งรหัสมา Server จะสุ่มให้ // ========================================== if (action === "reset_password") { let newPassword = body?.newPassword || body?.password || body?.extra?.newPassword || body?.extra?.password; // ถ้าไม่ได้ส่งมา ให้ระบบสร้างเอง if ( typeof newPassword !== "string" || newPassword.trim().length < 6 ) { newPassword = generatePassword(10); } const { data, error, } = await adminClient.auth.admin.updateUserById( targetId, { password: newPassword, }, ); if (error) { console.error( "reset password error:", error, ); return json( { ok: false, error: error.message || "รีเซ็ตรหัสผ่านไม่สำเร็จ", }, 400, ); } return json({ ok: true, message: "รีเซ็ตรหัสผ่านสำเร็จ", userId: data?.user?.id || targetId, // สำคัญ: // ส่งกลับให้ Frontend แสดงครั้งเดียว newPassword, name: data?.user?.user_metadata?.name || targetUser.user.user_metadata?.name || "", }); } // ========================================== // ลบบัญชี // ========================================== if (action === "delete_user") { const { error } = await adminClient.auth.admin.deleteUser( targetId, ); if (error) { console.error( "delete user error:", error, ); return json( { ok: false, error: error.message || "ลบบัญชีไม่สำเร็จ", }, 400, ); } return json({ ok: true, message: "ลบบัญชีสำเร็จ", userId: targetId, }); } return json( { ok: false, error: "ไม่รู้จัก action: " + body.action, }, 400, ); } catch (error) { console.error( "admin-users fatal error:", error, ); return json( { ok: false, error: error instanceof Error ? error.message : "เซิร์ฟเวอร์เกิดข้อผิดพลาด", }, 500, ); } });
